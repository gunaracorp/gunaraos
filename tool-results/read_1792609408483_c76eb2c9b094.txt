    75→function posisiBadgeCls(pos: number): { label: string; cls: string } {
    76→  if (pos >= 1 && pos <= 3)
    77→    return { label: `#${pos}`, cls: "bg-emerald-600 text-white border-emerald-600" };
    78→  if (pos >= 4 && pos <= 10)
    79→    return { label: `#${pos}`, cls: "bg-lime-500/20 text-lime-800 dark:text-lime-300 border-lime-500/50" };
    80→  if (pos >= 11 && pos <= 30)
    81→    return { label: `#${pos}`, cls: "bg-amber-400/25 text-amber-800 dark:text-amber-300 border-amber-400/50" };
    82→  if (pos >= 31 && pos <= 99)
    83→    return { label: `#${pos}`, cls: "bg-orange-300/25 text-orange-800 dark:text-orange-300 border-orange-400/50" };
    84→  return { label: "Tidak ada", cls: "bg-muted text-muted-foreground border-border" };
    85→}

   146→function DeltaCell({ pos, prev }: { pos: number; prev: number }) {
   147→  if (!prev || prev <= 0) {
   148→    return (
   149→      <span className="inline-flex items-center gap-1 text-xs text-muted-foreground">
   150→        <Minus className="size-3.5" aria-hidden /> —
   151→      </span>
   152→    );
   153→  }
   154→  if (pos < prev) {
   155→    return (
   156→      <span className="inline-flex items-center gap-1 text-xs font-semibold text-emerald-600 dark:text-emerald-400">
   157→        <ArrowUp className="size-3.5" aria-hidden /> {prev - pos}
   158→      </span>
   159→    );
   160→  }
   161→  if (pos > prev) {
   162→    return (
   163→      <span className="inline-flex items-center gap-1 text-xs font-semibold text-red-600 dark:text-red-400">
   164→        <ArrowDown className="size-3.5" aria-hidden /> {pos - prev}
   165→      </span>
   166→    );
   167→  }
   168→  return (
   169→    <span className="inline-flex items-center gap-1 text-xs text-muted-foreground">
   170→      <Minus className="size-3.5" aria-hidden /> Sama
   171→    </span>
   172→  );
   173→}

   195→export function RankTrackerSection() {
   196→  const { toast } = useToast();

   198→  const [projectId, setProjectId] = useState("");

   200→
   201→  // Form tambah keyword
   202→  const [fKeyword, setFKeyword] = useState("");
   203→  const [fPosisi, setFPosisi] = useState("");
   204→  const [fPrev, setFPrev] = useState("");
   205→  const [fVolume, setFVolume] = useState("");
   206→  const [fUrl, setFUrl] = useState("");
   207→  const [fPerangkat, setFPerangkat] = useState("desktop");
   208→  const [saving, setSaving] = useState(false);
   209→
   210→  const stats = useMemo(() => {
   211→    const items = ranks.items;
   212→    const positioned = items.filter((r) => r.posisi > 0);
   213→    const avg = positioned.length
   214→      ? positioned.reduce((s, r) => s + r.posisi, 0) / positioned.length
   215→      : 0;
   216→    return {
   217→      total: items.length,
   218→      avg,
   219→      top3: items.filter((r) => r.posisi >= 1 && r.posisi <= 3).length,
   220→      top10: items.filter((r) => r.posisi >= 1 && r.posisi <= 10).length,
   221→    };
   222→  }, [ranks.items]);
   223→
   224→  const sorted = useMemo(
   225→    () => [...ranks.items].sort((a, b) => (a.posisi || 101) - (b.posisi || 101)),
   226→    [ranks.items]
   227→  );
   228→
   229→  async function handleAdd() {
   230→    if (!projectId) {
   231→      toast({ title: "Pilih proyek dulu", description: "Pilih proyek SEO sebelum melacak keyword.", variant: "destructive" });
   232→      return;
   233→    }
   234→    const kw = fKeyword.trim();
   235→    if (!kw) {
   236→      toast({ title: "Keyword wajib diisi", description: "Tulis keyword yang ingin dilacak posisinya.", variant: "destructive" });
   237→      return;
   238→    }
   239→    setSaving(true);
   240→    try {
   241→      await ranks.create({
   242→        projectId,
   243→        keyword: kw,
   244→        posisi: Math.max(0, Math.min(100, Math.round(Number(fPosisi) || 0))),
   245→        posisiSebelumnya: Math.max(0, Math.round(Number(fPrev) || 0)),
   246→        volume: Math.max(0, Math.round(Number(fVolume) || 0)),
   247→        url: fUrl.trim(),
   248→        perangkat: fPerangkat,
   249→      });
   250→      toast({ title: "✅ Keyword dilacak", description: `“${kw}” masuk ke pemantauan posisi.` });
   251→      setFKeyword(""); setFPosisi(""); setFPrev(""); setFVolume(""); setFUrl("");
   252→    } catch (e) {
   253→      toast({ title: "Gagal menambah keyword", description: e instanceof Error ? e.message : "Coba lagi.", variant: "destructive" });
   254→    } finally {
   255→      setSaving(false);
   256→    }
   257→  }

   266→  }

   269→
   270→  return (
   271→    <div className="space-y-4">
   272→      <FadeUp>
   273→        <div className="rounded-xl border-2 border-emerald-800/20 bg-gradient-to-br from-emerald-50 to-amber-50 p-5 dark:border-emerald-700/30 dark:from-emerald-950/40 dark:to-amber-950/30">
   274→          <h2 className="text-lg font-extrabold tracking-tight">🎯 SEO RANK TRACKER — Pemantau Posisi Keyword</h2>
   275→          <p className="mt-1 max-w-3xl text-sm text-muted-foreground">
   276→            Pantau pergerakan posisi keyword di Google per perangkat. Hijau = naik, merah = turun.
   277→            Fokuskan tenaga mendorong keyword di posisi 11–30 ke halaman pertama.
   278→          </p>
   279→        </div>
   280→      </FadeUp>
   281→
   282→      {projects.loading ? (
   283→        <Card className="rounded-xl"><CardContent className="p-6 text-sm text-muted-foreground">Memuat proyek…</CardContent></Card>
   284→      ) : !hasProjects ? (
   285→        <Card className="rounded-xl">
   286→          <CardContent className="p-6">
   287→            <EmptyState
   288→              icon={FolderPlus}
   289→              title="Belum ada proyek SEO"
   290→              desc="Buat proyek dulu di Pusat Kendali SEO (SEO Hub) untuk mulai melacak posisi keyword."
   291→            />
   292→          </CardContent>
   293→        </Card>
   294→      ) : (
   295→        <>
   296→          <FadeUp delay={0.05}>
   297→            <Card className="rounded-xl">
   298→              <CardContent className="p-6">
   299→                <ProjectPicker
   300→                  id="rank-project-picker"
   301→                  projects={projects.items}
   302→                  loading={projects.loading}
   303→                  value={projectId}
   304→                  onChange={setProjectId}
   305→                />
   306→              </CardContent>
   307→            </Card>
   308→          </FadeUp>
   309→
   310→          {!projectId ? (
   311→            <Card className="rounded-xl">
   312→              <CardContent className="p-6">
   313→                <EmptyState
   314→                  icon={MousePointerClick}
   315→                  title="Pilih proyek untuk memuat data"
   316→                  desc="Posisi keyword, statistik, dan riwayat akan tampil setelah proyek dipilih."
   317→                />
   318→              </CardContent>
   319→            </Card>
   320→          ) : (
   321→            <>
   322→              {/* Statistik KPI */}
   323→              <FadeUp delay={0.1}>
   324→                <div className="grid grid-cols-2 gap-4 lg:grid-cols-4">

   330→                    sub="di luar posisi 100+"
   331→                    tone="text-amber-500"
   332→                  />
   333→                  <KpiCard icon={Crown} label="Top 3" value={fmtNum(stats.top3)} sub="posisi 1–3" tone="text-emerald-600 dark:text-emerald-400" />
   334→                  <KpiCard icon={TrendingUp} label="Top 10" value={fmtNum(stats.top10)} sub="posisi 1–10" tone="text-lime-600 dark:text-lime-400" />
   335→                </div>
   336→              </FadeUp>
   337→
   338→              {/* Form tambah keyword */}
   339→              <FadeUp delay={0.15}>
   340→                <Card className="rounded-xl">

   567→    const dofollow = items.filter((b) => b.dofollow).length;
   568→    const avgDa = total ? items.reduce((s, b) => s + (b.domainAuth || 0), 0) / total : 0;

   589→        targetUrl: bTarget.trim(),
   590→        anchorText: bAnchor.trim(),
   591→        dofollow: bDofollow,
   592→        kualitas: bKualitas,
   593→        status: "PENDING",
   594→        domainAuth: Math.max(0, Math.min(100, Math.round(Number(bDa) || 0))),
   595→      });

   623→            Kualitas backlink jauh lebih penting daripada kuantitas. Pantau atribut dofollow, otoritas domain (DA),

   635→              icon={FolderPlus}
