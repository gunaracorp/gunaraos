   127→function trunc(s: string, n: number): string {
   128→  return s.length > n ? s.slice(0, n).trimEnd() + "…" : s;
   129→}

   131→function hostOf(u: string): string {
   132→  const raw = u.trim();
   133→  if (!raw) return "";
   134→  try {
   135→    return new URL(raw.includes("://") ? raw : `https://${raw}`)
   136→      .hostname.replace(/^www\./, "");
   137→  } catch {
   138→    return raw.replace(/^https?:\/\//, "").replace(/^www\./, "").split("/")[0];
   139→  }
   140→}

   142→function counterClass(len: number, max: number, warn: number): string {
   143→  if (len === 0) return "text-muted-foreground";
   144→  if (len <= max) return "text-emerald-600 dark:text-emerald-400";
   145→  if (len <= warn) return "text-amber-600 dark:text-amber-400";
   146→  return "text-rose-500";
   147→}

   149→function fmtDate(iso?: string): string {
   150→  if (!iso) return "—";
   151→  try {
   152→    return new Date(iso).toLocaleDateString("id-ID", {
   153→      day: "numeric",
   154→      month: "short",
   155→      year: "numeric",
   156→      timeZone: "Asia/Jakarta",
   157→    });
   158→  } catch {
   159→    return "—";
   160→  }
   161→}

   168→/** Pecah URL menjadi rantai breadcrumb [Beranda, Segmen, …] */
   169→function buildCrumbs(u: string): { name: string; item: string }[] {
   170→  const raw = u.trim();
   171→  const base = raw.startsWith("http") ? raw : raw ? `https://${raw}` : "";
   172→  const crumbs: { name: string; item: string }[] = [];
   173→  try {
   174→    const url = new URL(base);
   175→    crumbs.push({ name: "Beranda", item: url.origin });
   176→    const segs = url.pathname.split("/").filter(Boolean);
   177→    let acc = url.origin;
   178→    for (const seg of segs) {
   179→      acc += `/${seg}`;
   180→      const label = decodeURIComponent(seg)
   181→        .replace(/[-_]+/g, " ")
   182→        .replace(/\.(html?|php|aspx)$/i, "");
   183→      crumbs.push({
   184→        name: label.charAt(0).toUpperCase() + label.slice(1),
   185→        item: acc,
   186→      });
   187→    }
   188→  } catch {
   189→    crumbs.push({ name: "Beranda", item: raw });
   190→  }
   191→  return crumbs;
   192→}

   477→      title: `Preset "${p.halaman}" dimuat`,
   478→      description: "Semua kolom terisi ulang — lanjutkan penyempurnaan.",
   479→    });
   480→  }
