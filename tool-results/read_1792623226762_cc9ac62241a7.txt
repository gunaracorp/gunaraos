    39→// ── Helpers format ──
    40→export const rupiah = (n: number | null | undefined) =>
    41→  "Rp " + (n || 0).toLocaleString("id-ID", { maximumFractionDigits: 0 });
    42→
    43→export const rupiahShort = (n: number | null | undefined) => {
    44→  const v = n || 0;
    45→  if (Math.abs(v) >= 1_000_000_000) return `Rp ${(v / 1_000_000_000).toLocaleString("id-ID", { maximumFractionDigits: 2 })} M`;
    46→  if (Math.abs(v) >= 1_000_000) return `Rp ${(v / 1_000_000).toLocaleString("id-ID", { maximumFractionDigits: 1 })} jt`;
    47→  if (Math.abs(v) >= 1_000) return `Rp ${(v / 1_000).toLocaleString("id-ID", { maximumFractionDigits: 0 })} rb`;
    48→  return rupiah(v);
    49→};
    50→
    51→/** Konversi Date → ISO tanggal (yyyy-mm-dd) memakai waktu LOKAL, bukan UTC.
    52→ *  Penting: pengguna berada di WIB (UTC+7) — toISOString() akan salah tanggal
    53→ *  antara jam 00.00–07.00 WIB. */
    54→export const isoFrom = (d: Date) => {
    55→  const y = d.getFullYear();
    56→  const m = String(d.getMonth() + 1).padStart(2, "0");
    57→  const dd = String(d.getDate()).padStart(2, "0");
    58→  return `${y}-${m}-${dd}`;
    59→};
    60→
    61→export const todayISO = () => isoFrom(new Date());
    62→
    63→export const fmtDate = (iso?: string | null) => {
    64→  if (!iso) return "—";
    65→  const d = new Date(iso);
    66→  if (isNaN(d.getTime())) return iso;
    67→  return d.toLocaleDateString("id-ID", { day: "numeric", month: "short", year: "numeric" });
    68→};
    69→
    70→export const dayName = (d: Date) =>
    71→  ["Minggu", "Senin", "Selasa", "Rabu", "Kamis", "Jumat", "Sabtu"][d.getDay()];
    72→
    73→export const fmtDateFull = (d: Date) =>
    74→  `${dayName(d)}, ${d.getDate()} ${d.toLocaleDateString("id-ID", { month: "long" })} ${d.getFullYear()}`;

    76→// ── Badge styling util ──
    77→export function statusTone(s: string): "default" | "green" | "amber" | "red" | "gray" {
    78→  const green = ["LUNAS", "SELESAI", "KLIEN", "RETAINER", "LIVE", "MASUK", "TERBIT", "AKTIF", "KUAT", "SANGAT KUAT"];
    79→  const amber = ["NEGOSIASI", "MENUNGGU", "TERJADWAL", "PIPIN", "CEK", "REDIRECT", "PLAN", "PROSPEK", "PENAWARAN", "KONTRAK", "PROSES", "LEAD", "DORMAN", "SEDANG", "UNDANGAN", "DITANGGUHKAN"];
    80→  const red = ["TERLAMBAT", "MACET", "TELAT", "MASA TENGGANG", "AKTIF-HUTANG", "GAGAL", "BATAL", "CHURN", "KOMPROMI", "TERKUNCI", "LEMAH", "DIJUAL", "DITUTUP", "NONAKTIF"];
    81→  if (green.includes(s)) return "green";
    82→  if (amber.includes(s)) return "amber";
    83→  if (red.includes(s)) return "red";
    84→  return "default";
    85→}

   756→export function BadgeMini({ text, tone }: { text: string; tone?: "green" | "amber" | "red" | "gray" }) {
   757→  return (
   758→    <span className={cn(
   759→      "inline-flex items-center rounded-full px-2 py-0.5 text-xs font-medium whitespace-nowrap",
   760→      tone === "green" && "bg-emerald-100 text-emerald-800 dark:bg-emerald-900/40 dark:text-emerald-300",
   761→      tone === "amber" && "bg-amber-100 text-amber-900 dark:bg-amber-900/40 dark:text-amber-300",
   762→      tone === "red" && "bg-rose-100 text-rose-800 dark:bg-rose-900/40 dark:text-rose-300",
   763→      (!tone || tone === "gray") && "bg-stone-200 text-stone-700 dark:bg-stone-800 dark:text-stone-300",
   764→    )}>{text}</span>
   765→  );
   766→}
